import {test,expect} from '@playwright/test';
for(const source of ['google','excel'])test(`actual ${source} Korean authoring choices apply, save and reopen as the same story`,async({page})=>{
 await page.goto('/?view=editor');await expect(page.getByTestId('save-status')).toHaveText('기기에 저장됨');
 await page.getByLabel('작품 파일 가져오기').setInputFiles(`docs/architecture/evidence/authoring-format/story-korean-${source}-edited.xlsx`);
 await page.getByRole('dialog',{name:'작품 가져오기 확인'}).getByRole('button',{name:'내 사본으로 가져오기'}).click();
 await expect(page.getByLabel('대사 / 해설',{exact:true})).toHaveValue('한글 양식에서 수정한 작품 🌳');
 await expect(page.getByLabel('글 종류',{exact:true})).toHaveValue('narration');await expect(page.getByTestId('save-status')).toHaveText('기기에 저장됨');
 await page.reload();await expect(page.getByLabel('글 종류',{exact:true})).toHaveValue('narration');await expect(page.getByLabel('대사 / 해설',{exact:true})).toHaveValue('한글 양식에서 수정한 작품 🌳');
});
