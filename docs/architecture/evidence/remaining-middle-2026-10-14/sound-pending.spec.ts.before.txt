import {test,expect} from '@playwright/test';
import {readFileSync} from 'node:fs';
for(const leaveCut of [false,true])test(`pending local registration preserves current edit scope ${leaveCut}`,async({page})=>{
 await page.goto('/?view=editor');await expect(page.getByTestId('save-status')).toHaveText('기기에 저장됨');await page.locator('[data-additional-tools]>summary').click();await page.getByRole('button',{name:'소리 편집',exact:true}).click();await page.getByLabel('컷 배경음 동작',{exact:true}).selectOption('play');const originalAsset=await page.getByLabel('컷 배경음',{exact:true}).inputValue();await page.evaluate(()=>{const original=File.prototype.arrayBuffer;File.prototype.arrayBuffer=function(){if(this.name!=='pending.wav')return original.call(this);return new Promise<ArrayBuffer>(resolve=>{(window as unknown as {releasePendingAudio:()=>void}).releasePendingAudio=()=>{void original.call(this).then(resolve);};});};});await page.getByLabel('컷 배경음 파일 등록',{exact:true}).setInputFiles({name:'pending.wav',mimeType:'audio/wav',buffer:readFileSync('apps/web/public/assets/audio/forest.wav')});await expect.poll(()=>page.evaluate(()=>typeof (window as unknown as {releasePendingAudio?:()=>void}).releasePendingAudio)).toBe('function');
 if(leaveCut){const position=page.getByLabel('현재 장과 컷 이동');await position.selectOption((await position.locator('option').nth(1).getAttribute('value'))!);await expect(page.getByLabel('컷 배경음 동작',{exact:true})).toHaveValue('default');}else{await page.getByLabel('조정할 소리',{exact:true}).selectOption('ambience');await page.getByLabel('컷 주변 소리 동작',{exact:true}).selectOption('play');}
 await page.evaluate(()=>(window as unknown as {releasePendingAudio:()=>void}).releasePendingAudio());await expect.poll(()=>page.evaluate(async()=>{return new Promise<boolean>((resolve,reject)=>{const request=indexedDB.open('knolstory-audio-assets-v1',2);request.onerror=()=>reject(request.error);request.onsuccess=()=>{const db=request.result;const tx=db.transaction('work-metadata','readonly');const records=tx.objectStore('work-metadata').getAll();tx.oncomplete=()=>{db.close();resolve(records.result.some((item:{name:string})=>item.name==='pending.wav'));};tx.onerror=()=>{db.close();reject(tx.error);};};});})).toBe(true);
 if(leaveCut){await expect(page.getByLabel('컷 배경음 동작',{exact:true})).toHaveValue('default');const position=page.getByLabel('현재 장과 컷 이동');await position.selectOption((await position.locator('option').nth(0).getAttribute('value'))!);await page.getByRole('button',{name:'글 편집',exact:true}).click();await page.locator('[data-additional-tools]>summary').click();await page.getByRole('button',{name:'소리 편집',exact:true}).click();await expect(page.getByLabel('컷 배경음',{exact:true}).locator('option').filter({hasText:'pending.wav'})).toHaveCount(1);await expect(page.getByLabel('컷 배경음',{exact:true})).toHaveValue(originalAsset);}else{await page.getByLabel('조정할 소리',{exact:true}).selectOption('music');await expect(page.getByLabel('컷 배경음',{exact:true})).toHaveValue(/^audio:custom:/);await page.getByLabel('조정할 소리',{exact:true}).selectOption('ambience');await expect(page.getByLabel('컷 주변 소리 동작',{exact:true})).toHaveValue('play');}
});
