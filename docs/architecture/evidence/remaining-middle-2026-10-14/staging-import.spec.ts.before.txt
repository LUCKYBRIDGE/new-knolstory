import {test,expect} from '@playwright/test';

 await open.click();dialog=page.getByRole('dialog',{name:'다른 컷 배치 가져오기'});await dialog.getByLabel('배치 원본 컷').selectOption(first);const box=(await dialog.boundingBox())!;expect(box.x).toBeGreaterThanOrEqual(0);expect(box.y).toBeGreaterThanOrEqual(0);expect(box.x+box.width).toBeLessThanOrEqual(size.width);expect(box.y+box.height).toBeLessThanOrEqual(size.height);

 await text.fill(before+'\n배치 적용 후에도 글을 수정합니다.');await page.getByRole('button',{name:'가져온 배치 되돌리기',exact:true}).click();await expect(page.getByLabel('인물 표시',{exact:true})).not.toHaveValue('none');await expect(text).toHaveValue(before+'\n배치 적용 후에도 글을 수정합니다.');
 await open.click();await page.getByRole('dialog').getByRole('button',{name:'선택한 배치 적용',exact:true}).click();await expect(page.getByTestId('save-status')).toHaveText('기기에 저장됨');await page.reload();await page.getByRole('button',{name:'배경·인물 편집',exact:true}).click();await expect(page.getByLabel('인물 표시',{exact:true})).toHaveValue('none');
 await page.locator('[data-work-menu]>summary').click();const pending=page.waitForEvent('download');await page.getByRole('button',{name:'작품 파일 내보내기',exact:true}).click();const download=await pending,document=parseAuthoringFile(readFileSync((await download.path())!,'utf8'));const line=document.project.lines.find((cut:{id:string})=>cut.id===second);expect(line.stageComposition).toMatchObject({leftActors:[],rightActors:[]});expect(line.text).toBe(before+'\n배치 적용 후에도 글을 수정합니다.');
